# ----------------------------------------------------------------------------
# Web framework + ASGI server
# ----------------------------------------------------------------------------
# We pick `granian` over `uvicorn` for two reasons:
#   1. Faster cold start (~100-200 ms vs uvicorn's ~600 ms), which
#      matters on Render's free tier where the worker is suspended
#      after 15 min of inactivity.
#   2. granian is implemented in Rust, so it tolerates the brief CPU
#      throttling Render applies to free instances without dropping
#      in-flight requests.
#
# The companion ASGI server is set in render.yaml's `startCommand`.
fastapi>=0.109.0
granian>=1.6.0
jinja2>=3.1.0
python-multipart>=0.0.6

# ----------------------------------------------------------------------------
# MCP server / client
# ----------------------------------------------------------------------------
# `fastmcp` is a thin wrapper that always pulls in `fastmcp-slim` (the
# runtime). `fastmcp-slim` ships with the bare client surface; to import
# `from fastmcp import FastMCP` (which our server module does) we also need
# the `[server]` extra, which brings in `mcp.types` and the rest of the
# server runtime. Without this, CI fails with:
#   ImportError: FastMCP server support is not installed.
#   Install `fastmcp` or `fastmcp-slim[server]`.
# Pin `mcp` explicitly so the resolver picks an SDK version with `mcp.types`.
fastmcp==3.4.5
fastmcp-slim[server]==3.4.5
mcp>=1.24.0,<2.0.0
